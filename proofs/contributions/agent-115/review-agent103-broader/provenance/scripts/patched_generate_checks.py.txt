#!/usr/bin/env python3
"""Generate bounded kernel checks for the frozen sixteen-job scalar certificate."""
import argparse
import json
from pathlib import Path
data_source = DATA_SOURCE

HERE = Path(__file__).resolve().parent


def checks(mode, identity, count, predicate, unfold, chunk):
    pieces=[]; proofs=[]
    for start in range(0,count,chunk):
        size=min(chunk,count-start)
        name=f'{mode}_{identity}_{start//chunk}'
        pieces.append(f'theorem {name} : ∀ i : Fin {size}, {predicate.replace("$u",f"({start}+i.val)")} := by\n  unfold {unfold}\n  decide\n')
        proofs.append((start,start+size,f'(checked_range (fun u => {predicate.replace("$u","u")}) {start} {size} {name})'))
    def join(items):
        if len(items)==1:return items[0]
        cut=len(items)//2
        lo,mid,left=join(items[:cut]);mid2,hi,right=join(items[cut:])
        assert mid==mid2
        expr=f'(join_ranges (fun u => {predicate.replace("$u","u")}) {lo} {mid} {hi} {left} {right})'
        return lo,hi,expr
    _,_,proof=join(proofs)
    pieces.append(f'theorem {mode}_{identity}_all : ∀ u, 0 ≤ u → u < {count} → {predicate.replace("$u","u")} :=\n  {proof}\n')
    return '\n'.join(pieces)


def main():
    ap=argparse.ArgumentParser(description=__doc__)
    ap.add_argument('--jobs',type=int,nargs='+',default=list(range(15)))
    ap.add_argument('--chunk',type=int,default=128)
    ap.add_argument('--output',type=Path,default=HERE/'generated')
    args=ap.parse_args();jobs=data_source.models()
    args.output.mkdir(exist_ok=True,parents=True)
    record=[]
    for identity in args.jobs:
        d=jobs[identity];n=len(d['meta']);p=d.get('lift',{}).get('parent')
        s=f'import BJobData{identity}\nimport RangeChecks\n'
        if p is not None:s+=f'import BJobData{p}\n'
        s+='namespace BroaderJobData\nopen ScalarFinite\nset_option maxHeartbeats 0\nset_option maxRecDepth 100000\n'
        s+=checks('vertex',identity,n,f'VertexRow ranking{identity} $u','VertexRow',args.chunk)
        s+=checks('edge',identity,n,f'∀ bit : Fin 2, EdgeRow data{identity} ranking{identity} $u bit.val','EdgeRow',args.chunk)
        s+=f'''theorem checked_rank{identity} : CheckedRank data{identity} ranking{identity} := by
  constructor
  · intro u hu
    exact vertex_{identity}_all u (Nat.zero_le u) hu
  · intro u hu bit hb
    exact edge_{identity}_all u (Nat.zero_le u) hu ⟨bit,by omega⟩
'''
        if p is not None:
            lift=d['lift'];selected=lift['selected'];m=len(lift['members']);pn=len(jobs[p]['meta'])
            s+=checks('coverage',identity,pn,f'CoverageRow data{p} component{p} {selected} {m} member{identity} memberIndex{identity} $u','CoverageRow',args.chunk)
            s+=checks('field',identity,n,f'FieldRow data{p} data{identity} member{identity} $u','FieldRow',args.chunk)
            s+=checks('lift',identity,n,f'∀ bit : Fin 2, LiftRow data{p} data{identity} component{p} {selected} member{identity} memberIndex{identity} $u bit.val','LiftRow',args.chunk)
            s+=f'''def lift_certificate{identity} : LiftCertificate data{p} data{identity} component{p} {selected} where
  members := member{identity}
  memberIndex := memberIndex{identity}
  count := {m}
  size_eq := rfl
  coverage u hu := coverage_{identity}_all u (Nat.zero_le u) hu
  fields u hu := field_{identity}_all u (Nat.zero_le u) hu
  transitions u hu bit hb := lift_{identity}_all u (Nat.zero_le u) hu ⟨bit,by omega⟩

theorem lift_complete{identity} : ActualPrefixLift.Realization
    (bounded (ActualPrefixLift.lift (ActualPrefixLift.select (graph data{p})
      (fun v => component{p} v={selected})))) (graph data{identity})
      (encode lift_certificate{identity}) := lift_realization lift_certificate{identity}
'''
        s+='end BroaderJobData\n'
        f=args.output/f'BJobChecks{identity}.lean';f.write_text(s)
        record.append({'job':identity,'vertices':n,'chunk_size':args.chunk,'check_source_bytes':len(s.encode())})
    (args.output/'check_generation.json').write_text(json.dumps({'jobs':record,'scope':'Generated kernel proof source; not accepted until Lean compilation succeeds.'},indent=2)+'\n')
    print(json.dumps({'jobs':args.jobs,'chunk_size':args.chunk,'total_check_source_bytes':sum(r['check_source_bytes'] for r in record)}))


if __name__=='__main__':main()
