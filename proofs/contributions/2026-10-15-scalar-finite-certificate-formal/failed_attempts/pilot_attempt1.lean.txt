namespace ScalarJobData
open ScalarFinite
set_option maxHeartbeats 0
set_option maxRecDepth 100000

theorem root_vertex_rows : ∀ u : Fin 7, VertexRow ranking0 u.val := by decide

theorem root_edge_rows : ∀ u : Fin 7, ∀ bit : Fin 2,
    EdgeRow data0 ranking0 u.val bit.val := by decide

theorem first_vertex_rows : ∀ u : Fin 133, VertexRow ranking1 u.val := by decide

theorem first_edge_rows : ∀ u : Fin 133, ∀ bit : Fin 2,
    EdgeRow data1 ranking1 u.val bit.val := by decide

theorem first_coverage : ∀ u : Fin 7,
    CoverageRow data0 component0 0 7 member1 memberIndex1 u.val := by decide

theorem first_fields : ∀ z : Fin 133, FieldRow data0 data1 member1 z.val := by decide

theorem first_transitions : ∀ z : Fin 133, ∀ bit : Fin 2,
    LiftRow data0 data1 component0 0 member1 memberIndex1 z.val bit.val := by decide

def first_lift_certificate : LiftCertificate data0 data1 component0 0 where
  members := member1
  memberIndex := memberIndex1
  count := 7
  size_eq := rfl
  coverage u hu := first_coverage ⟨u,hu⟩
  fields z hz := first_fields ⟨z,hz⟩
  transitions z hz bit hbit := first_transitions ⟨z,hz⟩ ⟨bit,by omega⟩

theorem first_lift_complete :
    ActualPrefixLift.Realization
      (bounded (ActualPrefixLift.lift (ActualPrefixLift.select (graph data0)
        (fun v => component0 v=0)))) (graph data1) (encode first_lift_certificate) :=
  lift_realization first_lift_certificate

end ScalarJobData
