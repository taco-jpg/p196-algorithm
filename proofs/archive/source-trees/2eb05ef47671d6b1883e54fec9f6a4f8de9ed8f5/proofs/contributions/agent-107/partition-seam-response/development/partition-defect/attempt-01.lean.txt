/-!
The cut-function seam defect and the consecutive-block seam mass are the
same quantity when the cuts describe the given blocks. Repeated boundaries,
empty blocks and wholly neutral blocks are included.
-/

/-- The two existing maximal initial-nine-run specifications agree. -/
theorem initialNineRun_iff_initialNines (q : Nat → Nat) (L ell : Nat) :
    InitialNineRun q L ell ↔ InitialNines q L ell := by
  constructor
  · intro h
    refine ⟨h.1, h.2.1, ?_⟩
    by_cases he : ell = L
    · exact Or.inl he
    · exact Or.inr (h.2.2 (by omega))
  · intro h
    refine ⟨h.1, h.2.1, ?_⟩
    intro hlt
    rcases h.2.2 with he | hn
    · omega
    · exact hn

theorem initialNinesLength_run (q : Nat → Nat) (L : Nat) :
    InitialNineRun q L (initialNinesLength q L) :=
  (initialNineRun_iff_initialNines q L _).mpr (initialNinesLength_spec q L)

/-- Identification of the recursive integer prefix sum with the list API. -/
theorem intSum_eq_list_sum (f : Nat → Int) (L : Nat) :
    intSum f L = ((List.range L).map f).sum := by
  induction L with
  | zero => simp [intSum]
  | succ L ih =>
      simp [intSum, List.range_succ, List.map_append, int_sum_append, ih]

/-- Split a prefix sum at any boundary, retaining an empty interval. -/
theorem intSum_add (f : Nat → Int) (a L : Nat) :
    intSum f (a + L) = intSum f a + intSum (fun i => f (a + i)) L := by
  induction L with
  | zero => simp [intSum]
  | succ L ih =>
      simp only [Nat.add_succ, intSum, ih]
      omega

/-- A genuine block contributes its actual incoming carry times its
computable initial-nine-run length to the recursive defect sum. -/
theorem intSum_block_defect (q : Nat → Nat) (cut : Nat → Bool) (a L : Nat)
    (ha : a = 0 ∨ cut a = true)
    (hinside : ∀ i, 0 < i → i < L → cut (a + i) = false)
    (hq : ∀ i, i < a + L → q i ≤ 18) :
    intSum (fun i => seamDefect q cut (a + i)) L =
      (carry q a : Int) * (initialNinesLength (fun i => q (a + i)) L : Int) := by
  rw [intSum_eq_list_sum]
  exact block_defect_sum q cut a L _ ha hinside hq
    (initialNinesLength_run (fun i => q (a + i)) L)

/-- The global cut-function defect sum is exactly the consecutive-partition
sum of actual incoming carry times initial-nine-run length. -/
theorem defect_sum_eq_seamMass (q boundary : Nat → Nat)
    (cut : Nat → Bool) (k : Nat)
    (hzero : boundary 0 = 0)
    (hmono : ∀ j, j < k → boundary j ≤ boundary (j + 1))
    (hq : ∀ i, i < boundary k → q i ≤ 18)
    (hstart : ∀ j, j < k → boundary j = 0 ∨ cut (boundary j) = true)
    (hinside : ∀ j, j < k → ∀ i, 0 < i → i < blockLength boundary j →
      cut (boundary j + i) = false) :
    intSum (seamDefect q cut) (boundary k) = (seamMass q boundary k : Int) := by
  revert hmono hq hstart hinside
  induction k with
  | zero =>
      intro _ _ _ _
      simp [hzero, intSum, seamMass, wordSum]
  | succ k ih =>
      intro hmono hq hstart hinside
      have hstep := hmono k (by omega)
      have hp := ih (fun j hj => hmono j (by omega))
        (fun i hi => hq i (by omega))
        (fun j hj => hstart j (by omega))
        (fun j hj => hinside j (by omega))
      have hlen : boundary k + blockLength boundary k = boundary (k + 1) := by
        unfold blockLength
        omega
      have hs := intSum_add (seamDefect q cut) (boundary k) (blockLength boundary k)
      rw [hlen] at hs
      have hb := intSum_block_defect q cut (boundary k) (blockLength boundary k)
        (hstart k (by omega)) (hinside k (by omega))
        (by intro i hi; exact hq i (by omega))
      have hb' : intSum (fun i => seamDefect q cut (boundary k + i))
          (blockLength boundary k) = (blockDeficit q boundary k : Int) := by
        simpa only [blockDeficit, blockWord, carryIn_zero_eq_carry,
          Int.natCast_mul] using hb
      change intSum (seamDefect q cut) (boundary (k + 1)) =
        ((seamMass q boundary k + blockDeficit q boundary k : Nat) : Int)
      rw [Int.natCast_add, hs, hp, hb']
